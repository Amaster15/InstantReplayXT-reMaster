cmake_minimum_required(VERSION 3.28)
project(InstantReplayXT VERSION 2.0.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
option(IRXT_REQUIRE_BASELINE "Require OBS 31.0.0 SDK for distributable baseline builds" OFF)
option(IRXT_BUILD_TESTS "Build Linux controller integration tests" OFF)
option(IRXT_PLUGIN_LAYOUT "Stage a self-contained per-plugin directory" ON)
# OBS packages use SameMajorVersion, so find_package(libobs 31) rejects OBS 32.
# Check the SDK's actual API macros in source instead of requesting that version.
find_package(libobs REQUIRED CONFIG)
find_package(obs-frontend-api REQUIRED CONFIG)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
if(IRXT_REQUIRE_BASELINE AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND Qt6_VERSION VERSION_GREATER_EQUAL 6.5)
  message(FATAL_ERROR "Linux baseline packages require Qt 6.4.x (Ubuntu 24.04). Disable IRXT_REQUIRE_BASELINE for local development with newer Qt.")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/build-versions.txt" "Qt=${Qt6_VERSION}\nCompiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\n")
add_library(InstantReplay MODULE module-entrance.cpp InstantReplay.cpp
  InstantReplay_callout.cpp InstantReplay_callout.h ui_InstantReplay_callback.h)
target_link_libraries(InstantReplay PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network)
target_compile_definitions(InstantReplay PRIVATE
  $<$<BOOL:${IRXT_REQUIRE_BASELINE}>:IRXT_REQUIRE_BASELINE>)
if(MSVC)
  target_compile_options(InstantReplay PRIVATE /W4 /utf-8)
  target_compile_definitions(InstantReplay PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows builds require x64; OBS 31+ does not support 32-bit Windows")
  endif()
else()
  target_compile_options(InstantReplay PRIVATE -Wall -Wextra)
endif()
set_target_properties(InstantReplay PROPERTIES PREFIX "")
if(APPLE)
  message(FATAL_ERROR "This port currently targets Windows x64 and Linux")
endif()
if(IRXT_PLUGIN_LAYOUT)
  install(TARGETS InstantReplay RUNTIME DESTINATION InstantReplay/bin/64bit
    LIBRARY DESTINATION InstantReplay/bin/64bit)
  install(FILES en-US.ini DESTINATION InstantReplay/data/locale)
elseif(WIN32)
  install(TARGETS InstantReplay RUNTIME DESTINATION obs-plugins/64bit
    LIBRARY DESTINATION obs-plugins/64bit)
  install(FILES en-US.ini DESTINATION data/obs-plugins/InstantReplay/locale)
else()
  include(GNUInstallDirs)
  install(TARGETS InstantReplay LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/obs-plugins)
  install(FILES en-US.ini DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/obs/obs-plugins/InstantReplay/locale)
endif()
install(FILES updater/install-after-obs.py updater/install-after-obs.ps1 DESTINATION InstantReplay/data/updater)
install(FILES README.md LICENSE DESTINATION .)
if(IRXT_BUILD_TESTS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The integration test frontend shim currently requires Linux")
  endif()
  enable_testing()
  add_executable(replay-controller-test tests/replay-controller-test.cpp)
  target_link_libraries(replay-controller-test PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network)
  target_compile_options(replay-controller-test PRIVATE -Wall -Wextra -UNDEBUG)
  add_test(NAME replay-controller COMMAND replay-controller-test)
  add_executable(update-info-test tests/update-info-test.cpp)
  target_link_libraries(update-info-test PRIVATE Qt6::Widgets Qt6::Network)
  target_compile_options(update-info-test PRIVATE -Wall -Wextra -UNDEBUG)
  add_test(NAME update-info COMMAND update-info-test)
  set_tests_properties(update-info PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(clip-panel-test tests/clip-panel-test.cpp)
  target_link_libraries(clip-panel-test PRIVATE Qt6::Widgets)
  target_compile_options(clip-panel-test PRIVATE -Wall -Wextra -UNDEBUG)
  add_test(NAME clip-panel COMMAND clip-panel-test)
  set_tests_properties(clip-panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(match-panel-test tests/match-panel-test.cpp)
  target_link_libraries(match-panel-test PRIVATE OBS::libobs Qt6::Widgets)
  target_compile_options(match-panel-test PRIVATE -Wall -Wextra -UNDEBUG)
  add_test(NAME match-panel COMMAND match-panel-test)
  set_tests_properties(match-panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(module-load-test tests/module-load-test.cpp)
  target_link_libraries(module-load-test PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network)
  target_compile_options(module-load-test PRIVATE -Wall -Wextra -UNDEBUG)
  set_target_properties(module-load-test PROPERTIES ENABLE_EXPORTS ON)
  add_dependencies(module-load-test InstantReplay)
  add_test(NAME module-load COMMAND module-load-test $<TARGET_FILE:InstantReplay>)
  set_tests_properties(module-load PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
